cmake_minimum_required(VERSION 3.16)
project(pikkabpe_firmware_core C)
set(CMAKE_C_STANDARD 99)
add_library(gimbal src/gimbal.c)
target_include_directories(gimbal PUBLIC include)
add_executable(test_gimbal tests/test_gimbal.c)
target_link_libraries(test_gimbal PRIVATE gimbal)
if(NOT MSVC)
  target_link_libraries(test_gimbal PRIVATE m)
  target_compile_options(gimbal PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME gimbal_core COMMAND test_gimbal)
